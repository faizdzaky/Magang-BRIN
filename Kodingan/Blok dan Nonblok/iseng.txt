#include <Arduino.h>

// Array pin untuk 6 LED
const int ledPins[] = {2, 3, 4, 5, 6, 7}; 
const int TOTAL_LEDS = 6;

// Pewaktu (Interval perpindahan lampu dalam milidetik)
unsigned long previousMillis = 0;
const long interval = 500; // Pindah setiap 0.5 detik (500 ms)

// Indeks LED yang sedang aktif (0 sampai 5)
int currentLedIndex = 0;

void setup() {
  // Mengatur semua pin LED sebagai OUTPUT menggunakan perulangan (loop)
  for (int i = 0; i < TOTAL_LEDS; i++) {
    pinMode(ledPins[i], OUTPUT);
    digitalWrite(ledPins[i], LOW); // Pastikan semua mati di awal
  }

  // Nyalakan lampu pertama di awal
  digitalWrite(ledPins[currentLedIndex], HIGH);
}

void loop() {
  unsigned long currentMillis = millis();

  // Cek apakah jeda waktu sudah terpenuhi
  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis; // Simpan waktu terakhir

    // 1. Matikan lampu yang sedang menyala saat ini
    digitalWrite(ledPins[currentLedIndex], LOW);

    // 2. Geser indeks ke lampu berikutnya
    // Modulo (%) membuat angka balik lagi ke 0 setelah mencapai 5
    currentLedIndex = (currentLedIndex + 1) % TOTAL_LEDS;

    // 3. Nyalakan lampu yang baru
    digitalWrite(ledPins[currentLedIndex], HIGH);
  }
}
